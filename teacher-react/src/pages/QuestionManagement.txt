// 题目类型接口
interface QuestionItem {
  id: number;
  type: 'image' | 'latex';
  subject: string;
  grade: string;
  year: string;
  question_type: string;
  folder_name?: string;
  path?: string;
  latex_code?: string;
  images?: Array<{name: string, url: string}>;
  is_selected: boolean;
}

// 题目统计数据接口
interface QuestionStats {
  id: number;
  type: 'image' | 'latex';
  starCount: number; 
  postCount: number;
  commentCount: number;
  likeCount: number;
  recommendCount: number;
  coinCount: number;
  favoriteCount: number;
  appleCount: number; // 学生总评分数
  studentCount: number; // 评分学生数
  averageRating: number; // 平均评分
  lastRefreshed: string;
  relatedPosts: Array<{
    id: number;
    title: string;
    author: string;
    commentCount: number;
    likeCount: number;
    coinCount: number;
    favoriteCount: number;
    created_at: string;
  }>;
}

const QuestionManagement: React.FC = () => {
  const { teacher, token } = useAuth();
  const { permissions } = usePermissions();
  
  // 筛选条件状态
  const [subject, setSubject] = useState<string>('');
  const [year, setYear] = useState<string>('');
  const [questionType, setQuestionType] = useState<string>('');
  const [grade, setGrade] = useState<string>('');
  const [selectionStatus, setSelectionStatus] = useState<'all' | 'selected' | 'unselected'>('all');
  const [searchKeyword, setSearchKeyword] = useState<string>('');
  
  // 数据状态
  const [loading, setLoading] = useState<boolean>(false);
  const [questions, setQuestions] = useState<QuestionItem[]>([]);
  const [filteredQuestions, setFilteredQuestions] = useState<QuestionItem[]>([]);
  const [selectedQuestionIds, setSelectedQuestionIds] = useState<number[]>([]);
  const [isAllSelected, setIsAllSelected] = useState<boolean>(false);
  const [recentlySelectedIds, setRecentlySelectedIds] = useState<number[]>([]);
  const [recentlyCanceledIds, setRecentlyCanceledIds] = useState<number[]>([]);
  
  // 分页状态
  const [currentPage, setCurrentPage] = useState<number>(1);
  const [totalQuestions, setTotalQuestions] = useState<number>(0);
  const [pageSize, setPageSize] = useState<number>(10);
  
  // 模式切换状态
  const [activeTab, setActiveTab] = useState<string>('all');
  
  // 调试模式状态
  const [debugMode, setDebugMode] = useState<boolean>(true);
  
  // 右侧导航状态
  const [rightTabActive, setRightTabActive] = useState<string>('selection');
  const [selectedDetailQuestion, setSelectedDetailQuestion] = useState<QuestionItem | null>(null);
  const [questionStats, setQuestionStats] = useState<QuestionStats | null>(null);
  const [statsLoading, setStatsLoading] = useState<boolean>(false);

  // 定时器引用
  const recentlySelectedTimeoutRef = useRef<number | null>(null);
  const recentlyCanceledTimeoutRef = useRef<number | null>(null);
  
  // 添加文件浏览器相关状态
  const [showFileExplorer, setShowFileExplorer] = useState<boolean>(false);
  
  // 初始化加载
  useEffect(() => {
    if (teacher?.id) {
      fetchQuestions();
    }
  }, [teacher, subject, year, questionType, grade, selectionStatus, currentPage, activeTab]);
  
  // 搜索关键词变化时过滤题目
  useEffect(() => {
    if (searchKeyword.trim() === '') {
      setFilteredQuestions(questions);
      return;
    }
    
    const keyword = searchKeyword.toLowerCase();
    const filtered = questions.filter(question => 
      (question.folder_name && question.folder_name.toLowerCase().includes(keyword)) ||
      question.subject.toLowerCase().includes(keyword) ||
      question.grade.toLowerCase().includes(keyword) ||
      question.year.toLowerCase().includes(keyword) ||
      question.question_type.toLowerCase().includes(keyword)
    );
    
    setFilteredQuestions(filtered);
  }, [searchKeyword, questions]);
  
  // 清理定时器
  useEffect(() => {
    return () => {
      if (recentlySelectedTimeoutRef.current) {
        window.clearTimeout(recentlySelectedTimeoutRef.current);
      }
      if (recentlyCanceledTimeoutRef.current) {
        window.clearTimeout(recentlyCanceledTimeoutRef.current);
      }
    };
  }, []);
  
  // 获取题目数据
  const fetchQuestions = async () => {
    if (!teacher?.id) return;
    
    setLoading(true);
    try {
      // 先获取题目的select状态，这样可以确保我们有准确的选中状态信息
      let selectStatusMap: Record<string, {
        question_id: number;
        question_type: string;
        select: string;
        is_selected: boolean;
      }> = {};
      
      try {
        const selectStatusResponse = await axios.get('http://localhost:3003/api/questions/select-status', {
          params: { teacherId: teacher.id },
          headers: { Authorization: `Bearer ${token}` }
        });
        
        if (debugMode) {
          console.log('数据库select状态信息:', selectStatusResponse.data);
        }
        
        selectStatusMap = selectStatusResponse.data.statusMap || {};
      } catch (error) {
        console.error('获取select状态失败:', error);
      }
      
      // 构建API请求参数，现在使用后端筛选
      const url = 'http://localhost:3003/api/questions';
      const params = {
        teacherId: teacher.id,
        subject,
        year,
        type: questionType,
        grade,
        selectionStatus, // 这个参数在后端用于筛选已选入/未选入的题目
        page: currentPage,
        limit: pageSize,
        tab: activeTab,
        // 添加随机参数，避免缓存
        _t: new Date().getTime()
      };
      
      if (debugMode) {
        console.log('获取题目, 参数:', params);
      }
      
      // 然后获取题目列表
      const response = await axios.get(url, {
        params,
        headers: { Authorization: `Bearer ${token}` }
      });
      
      if (debugMode) {
        console.log('题目数据响应:', response.data);
      }
      
      // 处理题目数据
      if (response.data.questions) {
        // 确保服务器返回的is_selected字段被正确解析为布尔值
        // 通过与selectStatusMap对比，确定真实的选中状态
        const processedQuestions = response.data.questions.map((q: any) => {
          const key = `${q.type}-${q.id}`;
          
          // 优先使用selectStatusMap中的select字段来决定选中状态
          const statusFromDB = selectStatusMap[key];
          
          // 只有当select字段为'1'时才视为已选入
          const isSelectedFromDB = statusFromDB && statusFromDB.select === '1';
          
          // 现在我们严格只使用数据库中的select字段
          // 如果没有找到对应的记录，则默认为未选入状态
          const finalIsSelected = isSelectedFromDB || false;
          
          if (debugMode) {
            console.log(`题目ID: ${q.id}, 类型: ${q.type}, 原始is_selected值: ${JSON.stringify(q.is_selected)}, 数据库select值: ${statusFromDB?.select}, 最终选中状态: ${finalIsSelected}`);
          }
          
          return {
            ...q,
            is_selected: finalIsSelected
          };
        });
        
        if (debugMode) {
          console.log('处理后的题目数据:', processedQuestions);
        }
        
        // 应用前端筛选（作为后端筛选的补充）
        let filteredBySelectStatus = processedQuestions;
        
        // 使用processedQuestions中的is_selected进行筛选
        if (selectionStatus === 'selected') {
          filteredBySelectStatus = processedQuestions.filter((q: any) => q.is_selected);
        } else if (selectionStatus === 'unselected') {
          filteredBySelectStatus = processedQuestions.filter((q: any) => !q.is_selected);
        }
        
        if (debugMode && filteredBySelectStatus.length !== processedQuestions.length) {
          console.log(`前端筛选: 从 ${processedQuestions.length} 条记录中筛选出 ${filteredBySelectStatus.length} 条`);
        }
        
        setQuestions(filteredBySelectStatus);
        setFilteredQuestions(filteredBySelectStatus);
        setTotalQuestions(response.data.total || processedQuestions.length);
        
        // 更新已选择的题目IDs (仅临时选择状态)
        // 这里只包含数据库中标记为已选入的题目ID
        const selectedIds = filteredBySelectStatus
          .filter((q: QuestionItem) => q.is_selected)
          .map((q: QuestionItem) => q.id);
        setSelectedQuestionIds(selectedIds);
      } else {
        setQuestions([]);
        setFilteredQuestions([]);
        setTotalQuestions(0);
        setSelectedQuestionIds([]);
      }
    } catch (error) {
      console.error('获取题目失败:', error);
      notification.error({
        message: '获取失败',
        description: '获取题目数据失败，请刷新页面重试',
        placement: 'topRight'
      });
    } finally {
      setLoading(false);
    }
  };
  
  // 处理筛选条件变更
  const handleFilterChange = () => {
    // 重置为第一页并重新获取题目
    setCurrentPage(1);
    
    if (debugMode) {
      console.log('筛选条件变更:', {
        subject,
        year,
        questionType,
        grade,
        selectionStatus
      });
    }
    
    fetchQuestions();
  };
  
  // 修改处理题目选择状态变更的函数
  const handleQuestionSelect = (questionId: number, checked: boolean) => {
    // 如果题目已经被选入，不允许通过复选框取消选择
    // 已选中的题目需要通过"取消选入"按钮来操作
    if (questions.find(q => q.id === questionId)?.is_selected) {
      return;
    }

    if (checked) {
      // 添加到已选择列表
      setSelectedQuestionIds(prev => [...prev, questionId]);
      
      // 添加到最近选择的列表，用于视觉反馈
      setRecentlySelectedIds(prev => [...prev, questionId]);
      
      // 设置定时器清除视觉反馈
      if (recentlySelectedTimeoutRef.current) {
        window.clearTimeout(recentlySelectedTimeoutRef.current);
      }
      recentlySelectedTimeoutRef.current = window.setTimeout(() => {
        setRecentlySelectedIds([]);
      }, 2000);
    } else {
      // 从已选择列表中移除
      setSelectedQuestionIds(prev => prev.filter(id => id !== questionId));
    }
    
    // 检查是否所有可选择的题目都已选中
    const selectableQuestions = filteredQuestions.filter(q => !q.is_selected);
    const newSelectedIds = checked 
      ? [...selectedQuestionIds, questionId]
      : selectedQuestionIds.filter(id => id !== questionId);
    
    setIsAllSelected(
      selectableQuestions.length > 0 && 
      selectableQuestions.every(q => newSelectedIds.includes(q.id))
    );
  };
  
  // 全选/取消全选当前页面的题目
  const handleSelectAllQuestions = (checked: boolean) => {
    if (checked) {
      // 获取当前页面显示的所有未选入的题目的ID
      const selectableQuestionIds = filteredQuestions
        .filter(q => !q.is_selected)
        .map(q => q.id);
      
      if (selectableQuestionIds.length === 0) {
        // 使用Modal替代message，避免React 19兼容性问题
        Modal.info({
          title: '提示',
          content: '当前页面没有可选择的题目',
          okText: '确定'
        });
        return;
      }
      
      // 将这些ID全部添加到已选择列表中
      setSelectedQuestionIds(prev => {
        const uniqueIds = new Set([...prev, ...selectableQuestionIds]);
        return Array.from(uniqueIds);
      });
      
      // 添加到最近选择的列表，用于视觉反馈
      setRecentlySelectedIds(selectableQuestionIds);
      
      // 设置定时器清除视觉反馈
      if (recentlySelectedTimeoutRef.current) {
        window.clearTimeout(recentlySelectedTimeoutRef.current);
      }
      recentlySelectedTimeoutRef.current = window.setTimeout(() => {
        setRecentlySelectedIds([]);
      }, 2000);
      
      setIsAllSelected(true);
      
      // 使用notification替代message，避免React 19兼容性问题
      notification.success({
        message: '批量选择成功',
        description: `已选择 ${selectableQuestionIds.length} 道题目`,
        placement: 'topRight'
      });
    } else {
      // 获取当前页面显示的所有未正式选入但被临时选中的题目的ID
      const currentPageQuestionIds = filteredQuestions
        .filter(q => !q.is_selected && selectedQuestionIds.includes(q.id))
        .map(q => q.id);
      
      if (currentPageQuestionIds.length === 0) {
        // 使用Modal替代message，避免React 19兼容性问题
        Modal.info({
          title: '提示',
          content: '没有可取消的临时选择',
          okText: '确定'
        });
        return;
      }
      
      // 从已选择列表中移除这些ID
      setSelectedQuestionIds(prev => 
        prev.filter(id => !currentPageQuestionIds.includes(id))
      );
      
      setIsAllSelected(false);
      
      // 使用notification替代message，避免React 19兼容性问题
      notification.info({
        message: '取消选择',
        description: `已取消选择 ${currentPageQuestionIds.length} 道题目`,
        placement: 'topRight'
      });
    }
  };
  
  // 批量选入题目
  const handleBatchSelect = () => {
    console.log('【按钮调试】批量选入按钮被点击');

    // 检查权限
    if (!permissions.canModifyQuestions) {
      notification.error({
        message: '权限不足',
        description: '您没有修改题库的权限',
        placement: 'topRight'
      });
      return;
    }

    if (!selectedQuestionIds.length) {
      notification.warning({
        message: '无法选入',
        description: '请至少选择一道题目',
        placement: 'topRight'
      });
      return;
    }
    
    console.log('【按钮调试】selectedQuestionIds:', selectedQuestionIds);
    
    // 获取所有"已临时选择"但未在数据库中标记为"已选入"的题目
    const tempSelectedQuestions = questions.filter(q => selectedQuestionIds.includes(q.id));
    console.log('【按钮调试】临时选择的题目:', tempSelectedQuestions.map(q => ({id: q.id, is_selected: q.is_selected, type: q.type})));
    
    const newQuestionIds = tempSelectedQuestions
      .filter(q => !q.is_selected)
      .map(q => q.id);
    
    console.log('【按钮调试】计算出需要选入的题目IDs:', newQuestionIds);
    
    if (newQuestionIds.length === 0) {
      notification.info({
        message: '提示',
        description: '所选题目已全部选入',
        placement: 'topRight'
      });
      return;
    }
    
    // 跳过确认对话框，直接调用提交函数进行测试
    console.log('【按钮调试】直接调用submitBatchSelect进行测试，不使用对话框');
    submitBatchSelect(newQuestionIds);
    
    /* 
    // 确认对话框方式，暂时注释以排查按钮问题
    Modal.confirm({
      title: '确认选入',
      content: `确定要选入这 ${newQuestionIds.length} 道题目吗？`,
      okText: '确定选入',
      cancelText: '取消',
      onOk: async () => {
        await submitBatchSelect(newQuestionIds);
      }
    });
    */
  };
  
  // 单独的提交函数，与UI交互分离
  const submitBatchSelect = async (questionIds: number[]): Promise<boolean> => {
    console.log('【提交调试】开始提交批量选入请求, questionIds:', questionIds);
    
    if (!questionIds.length || !teacher?.id) {
      console.log('【提交调试】参数无效:', { questionIds, teacherId: teacher?.id });
      notification.error({
        message: '操作失败',
        description: questionIds.length ? '教师信息未加载，请刷新页面重试' : '请至少选择一道题目',
        placement: 'topRight'
      });
      return false;
    }
    
    setLoading(true);
    console.log('【提交调试】设置loading状态为true');
    
    try {
      // 为每个ID找到对应的题目类型
      const questionDetails = questionIds.map(id => {
        const question = questions.find(q => q.id === id);
        return {
          id,
          type: question?.type || 'unknown'
        };
      });
      
      console.log('【提交调试】题目详情:', questionDetails);
      
      // 筛选出类型已知的题目
      const validQuestions = questionDetails.filter(q => q.type !== 'unknown');
      
      if (validQuestions.length === 0) {
        notification.error({
          message: '操作失败',
          description: '无法确定题目类型，请刷新页面重试',
          placement: 'topRight'
        });
        setLoading(false);
        return false;
      }
      
      // 准备请求数据，包含题目类型信息
      const requestData = {
        teacherId: teacher.id,
        questionIds, // 保留简单的ID数组格式
        questionTypes: validQuestions.map(q => q.type) // 添加类型信息数组
      };
      
      console.log('【提交调试】请求数据:', requestData);
      
      // 最简单的方式发送请求
      console.log('【提交调试】发送请求到:', 'http://localhost:3003/api/questions/select');
      const response = await axios.post(
        'http://localhost:3003/api/questions/select',
        requestData,
        {
          headers: { 
            'Authorization': `Bearer ${token}`,
            'Content-Type': 'application/json'
          }
        }
      );
      
      console.log('【提交调试】收到响应:', response.data);
      
      if (response.data.selectedCount > 0) {
        notification.success({
          message: '选入成功',
          description: `已成功选入 ${response.data.selectedCount} 道题目`,
          placement: 'topRight'
        });
        
        // 清除临时选择状态
        console.log('【提交调试】清除临时选择状态');
        setSelectedQuestionIds([]);
        setIsAllSelected(false);
        
        // 优化：在重新获取数据前，先更新本地状态以提供即时反馈
        console.log('【提交调试】更新本地状态');
        setQuestions(prev => {
          const updated = prev.map(q => 
            questionIds.includes(q.id) 
              ? { ...q, is_selected: true } 
              : q
          );
          console.log('【提交调试】更新后的题目状态示例:', updated.slice(0, 3));
          return updated;
        });
        
        // 更新临时筛选的题目
        setFilteredQuestions(prev => 
          prev.map(q => 
            questionIds.includes(q.id) 
              ? { ...q, is_selected: true } 
              : q
          )
        );
        
        // 强制刷新页面，确保显示最新状态
        console.log('【提交调试】设置延迟重新获取数据');
        setTimeout(() => {
          console.log('【提交调试】开始重新获取数据');
          fetchQuestions();
        }, 1500); // 增加延迟时间，确保后端数据已更新
        
        return true;
      } else {
        notification.info({
          message: '选入结果',
          description: `已选入 ${response.data.selectedCount}/${response.data.totalCount} 道题目`,
          placement: 'topRight'
        });
        console.log('【提交调试】选入结果为0，可能是题目已经选入或后端处理错误');
        return false;
      }
    } catch (error: any) {
      console.error('【提交调试】批量选入失败:', error);
      
      let errorMessage = '未知错误';
      if (error.response) {
        console.error('【提交调试】响应错误:', {
          status: error.response.status,
          data: error.response.data
        });
        errorMessage = error.response.data?.message || `服务器响应错误(${error.response.status})`;
      } else if (error.request) {
        console.error('【提交调试】请求错误 - 未收到响应');
        errorMessage = '服务器未响应，请检查网络连接';
      } else {
        console.error('【提交调试】其他错误:', error.message);
        errorMessage = error.message;
      }
      
      notification.error({
        message: '选入失败',
        description: `操作未完成: ${errorMessage}`,
        placement: 'topRight'
      });
      return false;
    } finally {
      console.log('【提交调试】设置loading状态为false');
      setLoading(false);
    }
  };
  
  // 取消选入题目
  const handleDeselectQuestion = (question: any) => {
    console.log('===================================');
    console.log('【前端】开始取消选择题目');
    console.log('【前端】时间:', new Date().toISOString());
    console.log('【前端】题目数据:', question);
    
    // 检查权限
    if (!permissions.canModifyQuestions) {
      notification.error({
        message: '权限不足',
        description: '您没有修改题库的权限',
        placement: 'topRight'
      });
      return;
    }
    
    // 由于已选入的题目卡片上只会显示"取消选入"按钮，所以不需要再次检查选中状态
    // 直接处理取消选入操作
    
    // 确认题目类型
    const questionType = question.type || question.question_type;
    if (!questionType) {

    if (value === undefined || value === null) return defaultValue;
    if (typeof value === 'number') return Math.floor(value); // 确保是整数
    if (typeof value === 'string') {
      const parsed = parseInt(value, 10);
      return isNaN(parsed) ? defaultValue : parsed;
    }
    return defaultValue;

  const safeParseFloat = (value: any, defaultValue: number = 0): number => {
    if (value === undefined || value === null) return defaultValue;
    if (typeof value === 'number') return value;
    if (typeof value === 'string') {
      const parsed = parseFloat(value);
      return isNaN(parsed) ? defaultValue : parsed;

    return defaultValue;

      return (
        <div style={{ textAlign: 'center', padding: '40px 0' }}>
          <Spin size="large" />
          <p>加载题目中...</p>
        </div>
      );
    }
    
    if (filteredQuestions.length === 0) {
      return (
        <div style={{ textAlign: 'center', padding: '40px 0' }}>
          <Empty 
            description={searchKeyword ? "没有找到匹配的题目" : "没有找到符合条件的题目"}
            image={Empty.PRESENTED_IMAGE_SIMPLE} 
          />
          {searchKeyword && (
            <Button 
              type="primary" 
              onClick={() => setSearchKeyword('')}
              style={{ marginTop: '16px' }}
            >
              清除搜索
            </Button>
          )}
        </div>
      );
    }
    
    return (
      <Row gutter={[16, 16]}>
        {filteredQuestions.map(question => {
          // 判断题目的状态
          const isSelected = selectedQuestionIds.includes(question.id);
          const isOfficiallySelected = question.is_selected;
          const isRecentlySelected = recentlySelectedIds.includes(question.id);
          const isRecentlyCanceled = recentlyCanceledIds.includes(question.id);
          
          if (debugMode) {
            // 仅在调试模式下输出当前题目的选中状态，帮助排查问题
            console.log(`题目ID: ${question.id}, 是否官方选中: ${isOfficiallySelected}, 原始is_selected值: ${JSON.stringify(question.is_selected)}`);
          }
          
          // 状态颜色和文本
          let statusColor = isOfficiallySelected ? "blue" : (isSelected ? "green" : "");
          let statusText = isOfficiallySelected ? "已选入" : (isSelected ? "已选择" : "");
          
          // 边框和阴影样式
          const borderStyle = isRecentlySelected ? '2px solid #52c41a' :
                              isRecentlyCanceled ? '2px solid #ff4d4f' :
                              isOfficiallySelected ? '1px solid #1890ff' : 
                              isSelected ? '1px solid #52c41a' : 
                              '1px solid #f0f0f0';
          
          const shadowStyle = isRecentlySelected ? '0 0 15px rgba(82, 196, 26, 0.8)' : 
                              isRecentlyCanceled ? '0 0 15px rgba(255, 77, 79, 0.8)' :
                              isOfficiallySelected ? '0 0 8px rgba(24, 144, 255, 0.5)' : 
                              isSelected ? '0 0 8px rgba(82, 196, 26, 0.5)' : 
                              '0 1px 2px rgba(0,0,0,0.1)';
          
          return (
            <Col span={12} key={question.id}>
              <Badge.Ribbon 
                text={statusText} 
                color={statusColor}
                style={{ display: statusText ? 'block' : 'none' }}
              >
                <Card
                  title={
                    <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
                      <Space>
                        <Text strong>{question.folder_name || `题目 ${question.id}`}</Text>
                        <Tag color={question.type === 'latex' ? 'blue' : 'green'}>
                          {question.type === 'latex' ? 'LaTeX' : '图片'}
                        </Tag>
                      </Space>
                      {/* 未选入的题目显示复选框供临时选择 */}
                      {/* {!isOfficiallySelected ? (
                        <Checkbox
                          checked={isSelected}
                          onChange={(e) => handleQuestionSelect(question.id, e.target.checked)}
                          style={{ 
                            transform: 'scale(1.2)',
                            '--ant-primary-color': '#52c41a'
                          } as React.CSSProperties}
                        />
                      ) : (
                        <Tag color="blue" style={{ margin: 0 }}>已选入</Tag>
                      )} */}
                    </div>
                  }
                  extra={
                    <Space>
                      <Tag color="orange">{question.subject}</Tag>
                      <Tag color="purple">{question.grade}</Tag>
                      <Tag>{question.year}</Tag>
                    </Space>
                  }
                  style={{ 
                    marginBottom: '16px', 
                    height: '100%',
                    boxShadow: shadowStyle,
                    border: borderStyle,
                    transition: 'all 0.3s ease'
                  }}
                  hoverable
                  actions={[
                    <Button 
                      type="text" 
                      icon={<InfoCircleOutlined />} 
                      onClick={() => handleSelectDetailQuestion(question)}
                    >
                      查看详情
                    </Button>,
                    isOfficiallySelected ? (
                      <Button 
                        danger
                        type="primary" 
                        size="small" 
                        icon={<DeleteOutlined />}
                        onClick={() => handleDeselectQuestion(question)}
                        disabled={!permissions.canModifyQuestions}
                      >
                        取消选入
                      </Button>
                    ) : (
                      <Space>
                        <Button 
                          type="primary" 
                          size="small" 
                          icon={<CheckCircleOutlined />}
                          onClick={() => submitBatchSelect([question.id])}
                          disabled={!permissions.canModifyQuestions}
                        >
                          选入
                        </Button>
                        <Checkbox
                          checked={isSelected}
                          onChange={(e) => handleQuestionSelect(question.id, e.target.checked)}
                          style={{ 
                            transform: 'scale(1.2)',
                            '--ant-primary-color': '#52c41a'
                          } as React.CSSProperties}
                        >{isSelected ? '已临时选择' : '临时选择'}
                        </Checkbox>
                      </Space>
                    )
                  ]}
                >
                  <div style={{ height: '220px', overflow: 'auto' }}>
                    {question.type === 'image' 
                      ? renderImageQuestion(question) 
                      : renderLatexQuestion(question)}
                  </div>
                </Card>
              </Badge.Ribbon>
            </Col>
          );
        })}
      </Row>
    );
  };
  
  // 切换调试模式
  const toggleDebugMode = () => {
    setDebugMode(!debugMode);
  };
  
  // 处理页码变更
  const handlePageChange = (page: number) => {
    setCurrentPage(page);
  };
  
  return (
    <Layout style={{ minHeight: '100vh', background: '#fff' }}>
      <Sider width={250} style={{ background: '#fff', padding: '16px', borderRight: '1px solid #f0f0f0' }}>
        <Title level={4}>题目筛选</Title>
        
        <Tabs
          activeKey={activeTab}
          onChange={(key) => {
            setActiveTab(key);
            setCurrentPage(1);
          }}
          items={[
            {
              key: 'all',
              label: '所有题目',
            },
            {
              key: 'image',
              label: '图片题目',
            },
            {
              key: 'latex',
              label: 'LaTeX题目',
            }
          ]}
        />
        
        <Divider />
        
        <div style={{ marginBottom: '16px' }}>
          <Text strong>学科:</Text>
          <Select
            style={{ width: '100%', marginTop: '8px' }}
            placeholder="选择学科"
            allowClear
            value={subject}
            onChange={(value) => {
              setSubject(value);
              // 重置后续所有筛选条件
              setGrade('');
              setYear('');
              setQuestionType('');
            }}
          >
            <Option value="线性代数">线性代数</Option>
            <Option value="数学分析">数学分析</Option>
            <Option value="英语">英语</Option>
            <Option value="大学物理">大学物理</Option>
            <Option value="电磁场">电磁场</Option>
          </Select>
        </div>
        
        <div style={{ marginBottom: '16px' }}>
          <Text strong>年级:</Text>
          <Select
            style={{ width: '100%', marginTop: '8px' }}
            placeholder="选择年级"
            allowClear
            value={grade}
            onChange={(value) => {
              setGrade(value);
              // 重置后续所有筛选条件
              setYear('');
              setQuestionType('');
            }}
            disabled={!subject} // 如果没有选择学科，禁用年级选择
          >
            <Option value="大一">大一</Option>
            <Option value="大二">大二</Option>
            <Option value="大三">大三</Option>
          </Select>
        </div>
        
        <div style={{ marginBottom: '16px' }}>
          <Text strong>年份:</Text>
          <Select
            style={{ width: '100%', marginTop: '8px' }}
            placeholder="选择年份"
            allowClear
            value={year}
            onChange={(value) => {
              setYear(value);
              // 重置后续所有筛选条件
              setQuestionType('');
            }}
            disabled={!grade} // 如果没有选择年级，禁用年份选择
          >
            <Option value="2019">2019</Option>
            <Option value="2020">2020</Option>
            <Option value="2021">2021</Option>
            <Option value="2022">2022</Option>
            <Option value="2023">2023</Option>
            <Option value="2024">2024</Option>
          </Select>
        </div>
        
        <div style={{ marginBottom: '16px' }}>
          <Text strong>类型:</Text>
          <Select
            style={{ width: '100%', marginTop: '8px' }}
            placeholder="选择类型"
            allowClear
            value={questionType}
            onChange={(value) => setQuestionType(value)}
            disabled={!year} // 如果没有选择年份，禁用类型选择
          >
            <Option value="A卷">A卷</Option>
            <Option value="B卷">B卷</Option>
          </Select>
        </div>
        
        <Divider />
        
        <div style={{ marginBottom: '16px' }}>
          <Text strong>筛选题目:</Text>
          <Radio.Group 
            onChange={(e) => setSelectionStatus(e.target.value)} 
            value={selectionStatus}
            style={{ marginTop: '8px', display: 'flex', flexDirection: 'column' }}
          >
            <Radio value="all">显示全部题目</Radio>
            <Radio value="selected">只显示已选入题目</Radio>
            <Radio value="unselected">只显示未选入题目</Radio>
          </Radio.Group>
        </div>
        
        <Button 
          type="primary" 
          icon={<FilterOutlined />} 
          style={{ marginRight: '8px', marginBottom: '16px' }}
          onClick={handleFilterChange}
        >
          应用筛选
        </Button>
        
        <Button 
          icon={<ReloadOutlined />}
          onClick={() => {
            setSubject('');
            setYear('');
            setQuestionType('');
            setGrade('');
            setSelectionStatus('all');
            setCurrentPage(1);
            fetchQuestions();
          }}
        >
          重置筛选
        </Button>
        
        {selectedQuestionIds.length > 0 && (
          <div style={{ marginTop: '16px', position: 'sticky', bottom: '20px' }}>
            <Button 
              type="primary" 
              block
              icon={<CheckCircleOutlined />}
              onClick={handleBatchSelect}
              style={{ 
                height: '40px', 
                fontSize: '16px',
                background: '#52c41a', 
                borderColor: '#52c41a',
                boxShadow: '0 2px 6px rgba(82, 196, 26, 0.5)',
                marginBottom: '8px'
              }}
            >
              批量选入 ({selectedQuestionIds.filter(id => 
                !questions.find(q => q.id === id)?.is_selected
              ).length}题)
            </Button>
            
            <Button 
              danger
              block
              icon={<DeleteOutlined />}
              onClick={handleBatchCancelSelection}
              style={{ height: '40px' }}
              disabled={selectedQuestionIds.filter(id => !questions.find(q => q.id === id)?.is_selected).length === 0}
            >
              取消临时选择
            </Button>
          </div>
        )}
      </Sider>
      
      <Content style={{ padding: '16px 24px',minWidth: '70%', minHeight: 280, display: rightTabActive === 'selection' ? 'block' : 'none' }}>
        <div style={{ marginBottom: '16px', display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
          <Title level={4}>题目列表</Title>
          <Space size="middle">
            <Input.Search 
              placeholder="搜索题目名称、学科、年级..." 
              allowClear
              style={{ width: 250 }}
              value={searchKeyword}
              onChange={(e) => setSearchKeyword(e.target.value)}
              onSearch={(value) => setSearchKeyword(value)}
            />
            <Space>
              <Checkbox 
                checked={isAllSelected}
                onChange={(e) => handleSelectAllQuestions(e.target.checked)}
                disabled={filteredQuestions.filter(q => !q.is_selected).length === 0}
              >
                全选本页
              </Checkbox>
              {/* 仅在"已选入"或"全部"筛选状态下显示批量取消选入按钮 */}
              {(selectionStatus === 'selected' || selectionStatus === 'all') && 
                filteredQuestions.filter(q => q.is_selected).length > 0 && (
                <Button 
                  type="primary"
                  danger
                  size="small"
                  icon={<DeleteOutlined />}
                  loading={loading}
                  onClick={handleBatchDeselect}
                >
                  批量取消选入
                </Button>
              )}
            </Space>
          </Space>
        </div>
        
        <div style={{ marginBottom: '16px', display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
          {/* 左侧统计信息 */}
          <Space wrap>
            <Tag color="cyan">总计: {totalQuestions} 题</Tag>
            <Tag color="green">待选入: {selectedQuestionIds.filter(id => 
              !questions.find(q => q.id === id)?.is_selected
            ).length} 题</Tag>
            <Tag color="blue">已选入: {questions.filter(q => q.is_selected).length} 题</Tag>
            {searchKeyword && (
              <Tag color="purple">搜索结果: {filteredQuestions.length} 题</Tag>
            )}
          </Space>
          
          {/* 移除重复的批量操作按钮，避免UI上的冗余 */}
        </div>
        
        {renderQuestionList()}
        
        {/* 分页组件 */}
        {totalQuestions > 0 && (
          <div style={{ marginTop: '20px', textAlign: 'right' }}>
            <Pagination 
              current={currentPage}
              total={totalQuestions}
              pageSize={pageSize}
              onChange={handlePageChange}
              showSizeChanger={false}
              // showTotal={(total) => `共 ${total} 题`}
            />
          </div>
        )}
      </Content>
      
      {/* 详情内容区域 */}
      <Content style={{ padding: '16px 24px', minHeight: 280, display: rightTabActive === 'details' ? 'block' : 'none' }}>
        {renderQuestionDetails()}
      </Content>
      
      {/* 文件浏览器内容区域 */}
      <Content style={{ padding: '16px 24px', minHeight: 280, display: rightTabActive === 'files' ? 'block' : 'none' }}>
        <FileExplorer />
      </Content>
      
      {/* LaTeX浏览器内容区域 */}
      <Content style={{ padding: '16px 24px', minHeight: 280, display: rightTabActive === 'latex' ? 'block' : 'none' }}>
        <LatexBrowser />
      </Content>
      
      {/* 右侧导航栏 */}
      <Sider width={80} theme="light" style={{ background: '#fff', borderLeft: '1px solid #f0f0f0' }}>
        <div style={{ display: 'flex', flexDirection: 'column', alignItems: 'center', marginTop: '20px' }}>
          <Tooltip title="题目列表" placement="left">
            <Button 
              type={rightTabActive === 'selection' ? 'primary' : 'default'}
              shape="circle" 
              icon={<UnorderedListOutlined />} 
              size="large"
              style={{ marginBottom: '20px' }}
              onClick={() => setRightTabActive('selection')}
            />
          </Tooltip>
          <Tooltip title="题目详情" placement="left">
            <Button 
              type={rightTabActive === 'details' ? 'primary' : 'default'}
              shape="circle" 
              icon={<InfoCircleOutlined />} 
              size="large"
              style={{ marginBottom: '20px' }}
              onClick={() => setRightTabActive('details')}
              disabled={!selectedDetailQuestion}
            />
          </Tooltip>
          <Tooltip title="文件浏览器" placement="left">
            <Button 
              type={rightTabActive === 'files' ? 'primary' : 'default'}
              shape="circle" 
              icon={<FolderOutlined />} 
              size="large"
              style={{ marginBottom: '20px' }}
              onClick={() => setRightTabActive('files')}
            />
          </Tooltip>
          <Tooltip title="LaTeX浏览器" placement="left">
            <Button 
              type={rightTabActive === 'latex' ? 'primary' : 'default'}
              shape="circle" 
              icon={<CodeOutlined />} 
              size="large"
              onClick={() => setRightTabActive('latex')}
            />
          </Tooltip>
        </div>
      </Sider>
      
      {/* 调试模式开关 */}
      {debugMode && (
        <div 
          style={{ 
            position: 'fixed', 
            bottom: '10px', 
            right: '10px', 
            background: '#f0f0f0', 
            padding: '4px 8px', 
            borderRadius: '4px',
            cursor: 'pointer',
            zIndex: 1000
          }}
          onClick={toggleDebugMode}
        >
          调试模式: 开启
        </div>
      )}
    </Layout>
  );
};

export default QuestionManagement;
